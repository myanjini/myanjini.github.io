  2. 저장소, 서비스, 로그인 API

    app/repositories/login_failures.py
    """로그인 실패 기록 저장소"""
from datetime import datetime

from sqlalchemy import delete, func, select
from sqlalchemy.orm import Session

from app.models import LoginFailure


def count_since(db: Session, since: datetime, *, email: str | None = None, ip: str | None = None) -> int:
    stmt = select(func.count()).select_from(LoginFailure).where(LoginFailure.created_at > since)
    if email is not None:
        stmt = stmt.where(LoginFailure.email == email)
    if ip is not None:
        stmt = stmt.where(LoginFailure.ip == ip)
    return db.scalar(stmt)


def add(db: Session, email: str, ip: str, at: datetime) -> None:
    db.add(LoginFailure(email=email, ip=ip, created_at=at))


def clear(db: Session, email: str) -> None:
    db.execute(delete(LoginFailure).where(LoginFailure.email == email))

  4. 비밀번호 규칙

    app/schemas.py
    # 실제 서비스에서는 유출된 비밀번호 목록 수십만 개를 씁니다. 여기서는 예시로 몇 개만 둡니다.
COMMON_PASSWORDS = {
    "password1234", "1234567890", "qwerty1234", "qwertyuiop", "iloveyou12",
    "admin12345", "abcd123456", "1q2w3e4r5t", "password12", "welcome123",
}
    app/schemas.py
    class UserCreate(InputModel):
    """회원가입에 쓰는 입력 모델"""
    email: EmailStr = Field(examples=["user1@example.com"])
    password: str = Field(min_length=10, max_length=64, examples=["practice-pass-1"])

    @model_validator(mode="after")
    def check_password_rules(self):
        password = self.password.lower()
        email_id = self.email.split("@")[0].lower()
        if password in COMMON_PASSWORDS:
            raise ValueError("너무 흔한 비밀번호입니다.")
        if len(email_id) >= 3 and email_id in password:
            raise ValueError("비밀번호에 이메일 아이디를 넣을 수 없습니다.")
        if len(set(password)) < 4:
            raise ValueError("같은 글자를 반복한 비밀번호는 쓸 수 없습니다.")
        return self

    가져오기에 model_validator를 추가합니다. 비밀번호와 이메일을 함께 봐야 하므로 필드 하나가 아니라 모델 전체를 검사하는 model_validator(mode="after")를 씁니다. 규칙에 어긋나면 ValueError를 내고, FastAPI가 422로 바꿔 줍니다.

    비밀번호(이메일 mybook@example.com)기대하는 결과내 결과
    short-pw1422, at least 10 characters
    password1234422, 너무 흔한 비밀번호입니다.
    aaaaaaaaaaaa422, 같은 글자를 반복한 비밀번호는 쓸 수 없습니다.
    mybook-is-good422, 비밀번호에 이메일 아이디를 넣을 수 없습니다.
    correct-horse-battery201

    이미 가입한 사용자의 비밀번호는 바뀌지 않습니다. 실제 서비스에서는 다음 로그인 때 새 규칙에 맞게 바꾸도록 안내합니다.

    tests/test_attacks.py
    def login(client, email, password):
    return client.post("/auth/token", data={"username": email, "password": password})


def test_login_is_locked_after_five_failures(client, make_user):
    make_user("user1@example.com")
    for _ in range(5):
        assert login(client, "user1@example.com", "wrong-pass-0").status_code == 401
    response = login(client, "user1@example.com", "test-pass-123")  # 맞는 비밀번호여도 잠깁니다.
    assert response.status_code == 429
    assert response.headers["Retry-After"] == "900"


def test_successful_login_resets_failures(client, make_user):
    make_user("user1@example.com")
    for _ in range(4):
        login(client, "user1@example.com", "wrong-pass-0")
    assert login(client, "user1@example.com", "test-pass-123").status_code == 200
    for _ in range(4):
        login(client, "user1@example.com", "wrong-pass-0")
    assert login(client, "user1@example.com", "test-pass-123").status_code == 200


@pytest.mark.parametrize("password", ["short-pw1", "password1234", "aaaaaaaaaaaa", "mybook-mybook-1"])
def test_weak_password_is_rejected(client, password):
    response = client.post("/auth/signup", json={"email": "mybook@example.com", "password": password})
    assert response.status_code == 422


def make_token(payload: dict, key: str = TEST_KEY, algorithm: str = "HS256") -> str:
    return jwt.encode(payload, key, algorithm=algorithm)


@pytest.mark.parametrize("token", [
    make_token({"sub": "1", "exp": datetime.now(timezone.utc) - timedelta(minutes=1)}),    # 만료
    make_token({"sub": "1"}),                                                               # exp 없음
    make_token({"sub": "1", "exp": datetime.now(timezone.utc) + timedelta(minutes=5)}, "other-key-" * 4),  # 다른 키
    make_token({"sub": "1", "exp": datetime.now(timezone.utc) + timedelta(minutes=5)}, None, "none"),     # 서명 없음
])
def test_invalid_tokens_are_rejected(client, make_user, token):
    make_user("user1@example.com")
    response = client.get("/auth/me", headers={"Authorization": f"Bearer {token}"})
    assert response.status_code == 401
