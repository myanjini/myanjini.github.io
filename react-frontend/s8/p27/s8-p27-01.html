토큰 부착과 자동 재발급

파트 목표axios 인터셉터로 모든 요청에 토큰을 붙이고, 만료 시 한 번만 재발급해 원래 요청을 다시 보낼 수 있습니다.

S6의 client.js에 요청 인터셉터와 재발급 로직을 추가합니다. 파일 전체를 다음과 같이 바꿉니다.

src/api/client.js
import axios from 'axios';
import { toApiError, ApiError, NETWORK_ERROR_MESSAGE } from './errors';
import { tokenStore } from '../auth/tokenStore';

export const API_BASE_URL = import.meta.env.VITE_API_BASE_URL;

export const api = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
});

// 요청 인터셉터: 액세스 토큰이 있으면 Authorization 헤더에 붙입니다.
api.interceptors.request.use((config) => {
  const token = tokenStore.getAccess();
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// 이 경로들은 401이 나와도 재발급을 시도하지 않습니다.
const NO_RETRY_PATHS = ['/auth/token', '/auth/refresh', '/auth/signup', '/auth/logout'];

// 리프레시 요청이 동시에 여러 번 나가지 않도록 하나의 Promise를 공유합니다.
let refreshPromise = null;

async function refreshAccessToken() {
  const refreshToken = tokenStore.getRefresh();
  if (!refreshToken) throw new Error('NO_REFRESH_TOKEN');
  // 인터셉터가 없는 기본 axios로 보내 무한 반복을 막습니다.
  const { data } = await axios.post(`${API_BASE_URL}/auth/refresh`, {
    refresh_token: refreshToken,
  });
  tokenStore.save(data);
  return data.access_token;
}

// 재발급에 실패했을 때 로그인 상태를 지우는 일을 AuthProvider에 맡기기 위한 연결 고리입니다.
let onAuthExpired = () => {};
export function setOnAuthExpired(handler) {
  onAuthExpired = handler;
}

// 응답 인터셉터: 401이면 한 번만 재발급한 뒤 원래 요청을 다시 보냅니다.
api.interceptors.response.use(
  (response) => response,
  async (error) => {
    // 화면을 떠나 취소한 요청은 오류가 아니므로 그대로 넘깁니다.
    if (axios.isCancel(error)) throw error;

    const { config, response } = error;

    if (!response) {
      throw new ApiError({ status: 0, code: 'NETWORK_ERROR', message: NETWORK_ERROR_MESSAGE });
    }

    const canRetry = !NO_RETRY_PATHS.includes(config.url) && !config._retried;
    if (response.status === 401 && canRetry && tokenStore.getRefresh()) {
      config._retried = true;
      try {
        refreshPromise ??= refreshAccessToken().finally(() => {
          refreshPromise = null;
        });
        await refreshPromise;
        return api(config); // 새 토큰으로 원래 요청을 다시 보냅니다.
      } catch {
        tokenStore.clear();
        onAuthExpired();
      }
    }

    throw toApiError(response.status, response.data, response.headers);
  },
);

// 백엔드가 돌려준 /covers/... 같은 경로를 화면에서 열 수 있는 주소로 바꿉니다.
export function toAssetUrl(path) {
  if (!path) return null;
  return `${API_BASE_URL}${path}`;
}

이 코드에서 주의 깊게 볼 부분은 네 가지입니다.

1) 한 번만 재시도합니다. 재발급 후 다시 보낸 요청에도 401이 오면 무한히 반복할 수 있습니다. 요청 설정 객체에 _retried 표시를 남겨 두 번째 401은 그대로 오류로 넘깁니다.

2) 재발급 요청은 인터셉터를 거치지 않습니다. 재발급 요청을 api 인스턴스로 보내면, 재발급 자체가 401일 때 다시 재발급을 시도하는 순환이 생깁니다. 그래서 인터셉터가 없는 기본 axios.post를 사용합니다. 같은 이유로 로그인, 회원가입, 로그아웃, 재발급 경로는 NO_RETRY_PATHS로 제외합니다. 로그인 실패(401)는 "비밀번호가 틀렸다"는 뜻이지 토큰 만료가 아니기 때문입니다.

3) 동시에 여러 요청이 401을 받아도 재발급은 한 번만 합니다. 대시보드처럼 여러 요청을 동시에 보내는 화면에서 토큰이 만료되면 401이 거의 동시에 여러 개 도착합니다. 각각 재발급을 요청하면, 백엔드가 재발급 때마다 리프레시 토큰을 새로 바꾸므로 먼저 도착한 재발급만 성공하고 나머지는 이미 바뀐 토큰을 들고 가서 실패합니다. 이를 막기 위해 진행 중인 재발급 Promise(refreshPromise)를 공유합니다. ??= 연산자는 값이 비어 있을 때만 새로 대입하므로, 첫 401만 재발급을 시작하고 나머지는 같은 Promise를 기다립니다. 재발급이 끝나면 finally에서 비워 다음 만료에 대비합니다.

4) 재발급 실패를 화면에 알립니다. 리프레시 토큰까지 만료되었으면 토큰을 지우고 onAuthExpired를 호출합니다. 이 함수는 S8-P28-01의 AuthProvider가 등록하며, 로그인 상태를 비워 화면이 로그아웃 상태로 바뀌게 합니다. client.js는 React 밖의 일반 모듈이라 React의 state를 직접 바꿀 수 없으므로, 이렇게 함수를 등록받는 방식으로 연결합니다.
