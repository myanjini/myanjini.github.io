토큰 보관소와 인증 API

이 과정은 액세스 토큰을 JavaScript 변수(메모리)에만 두고, 리프레시 토큰은 sessionStorage에 둡니다.

src/auth/tokenStore.js
// 토큰 보관소
// - 액세스 토큰: 메모리(모듈 변수)에만 둡니다. 새로고침하면 사라집니다.
// - 리프레시 토큰: sessionStorage에 둡니다. 탭을 닫으면 사라집니다.
// 저장 위치를 이렇게 고른 이유는 교안의 "프론트엔드 개발 보안" 단원에서 다룹니다.
const REFRESH_KEY = 'book-admin.refresh';

let accessToken = null;

export const tokenStore = {
  getAccess: () => accessToken,
  getRefresh: () => sessionStorage.getItem(REFRESH_KEY),
  save({ access_token, refresh_token }) {
    accessToken = access_token;
    if (refresh_token) sessionStorage.setItem(REFRESH_KEY, refresh_token);
  },
  clear() {
    accessToken = null;
    sessionStorage.removeItem(REFRESH_KEY);
  },
};

메모리에 둔 액세스 토큰은 새로고침하면 사라지지만, 리프레시 토큰으로 바로 다시 받을 수 있습니다. sessionStorage는 탭 단위로 분리되며 탭을 닫으면 지워집니다. 저장 위치마다 장단점이 있고 정답이 하나로 정해져 있지 않으므로, 각 방식의 위험과 이 선택의 근거는 S12에서 따로 비교합니다. 여기서는 "토큰을 어디에 두든 화면 코드는 tokenStore만 통해 접근한다"는 구조에 집중합니다. 저장 위치를 바꾸더라도 이 파일 하나만 고치면 됩니다.

src/api/auth.js
import { api } from './client';

export async function signup(email, password) {
  const { data } = await api.post('/auth/signup', { email, password });
  return data;
}

// OAuth2 Password Flow: JSON이 아니라 폼 형식으로 보내야 합니다.
export async function login(email, password) {
  const form = new URLSearchParams();
  form.append('username', email); // 이 백엔드는 username 칸에 이메일을 받습니다.
  form.append('password', password);
  const { data } = await api.post('/auth/token', form);
  return data; // { access_token, refresh_token, token_type }
}

export async function fetchMe() {
  const { data } = await api.get('/auth/me');
  return data; // { id, email, role }
}

export async function logout(refreshToken) {
  await api.post('/auth/logout', { refresh_token: refreshToken });
}

로그인은 S6에서 배운 대로 URLSearchParams로 폼 형식을 만들고, username 칸에 이메일을 넣습니다. axios는 URLSearchParams를 받으면 Content-Type: application/x-www-form-urlencoded를 자동으로 지정합니다.
