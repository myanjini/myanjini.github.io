대여와 반납 API: src/api/loans.js

src/api/loans.js
import { api } from './client';

export async function borrowBook(bookId) {
  const { data } = await api.post('/loans', { book_id: bookId });
  return data;
}

export async function returnLoan(loanId) {
  const { data } = await api.post(`/loans/${loanId}/return`);
  return data;
}

export async function listMyLoans(signal) {
  const { data } = await api.get('/loans/me', { signal });
  return data;
}

대여 요청 본문에는 book_id만 있습니다. 빌리는 사람은 백엔드가 토큰에서 알아냅니다. 백엔드 과정 S6-P21의 "흔한 실수" 가운데 하나가 본문으로 user_id를 받는 것이었는데, 프론트엔드도 같은 원칙을 따릅니다. 사용자를 식별하는 값을 요청 본문으로 보내지 않습니다.
