폼과 제어 컴포넌트

파트 목표입력 칸을 state와 연결한 제어 컴포넌트로 폼을 만들고, 입력값을 검증하고 변환할 수 있습니다.

입력 칸의 값을 state와 연결해 React가 항상 값을 쥐고 있게 만든 것을 제어 컴포넌트라고 합니다. 입력값을 검증하거나, 서버에 보낼 값으로 변환하거나, 오류 메시지를 칸마다 보여 줄 때 이 방식이 편리합니다.

function SearchBox({ onSearch }) {
  const [keyword, setKeyword] = useState('');

  function handleSubmit(event) {
    event.preventDefault(); // 폼 제출로 페이지가 새로고침되는 기본 동작을 막습니다.
    onSearch(keyword.trim());
  }

  return (
    <form onSubmit={handleSubmit}>
      <input value={keyword} onChange={(e) => setKeyword(e.target.value)} />
      <button type="submit">검색</button>
    </form>
  );
}

입력 칸의 값은 항상 문자열이라는 점을 기억합니다. type="number"인 칸도 e.target.value는 "18000" 같은 문자열입니다. API로 보낼 때 Number()로 바꿔야 백엔드가 정수로 받습니다. S9의 도서 등록 폼에서 이 변환을 함수로 분리합니다.
