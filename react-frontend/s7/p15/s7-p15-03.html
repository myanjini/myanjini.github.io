인증 API: src/api/auth.js

src/api/auth.js
import { api } from './client';

export async function signup(email, password) {
  const { data } = await api.post('/auth/signup', { email, password });
  return data;
}

// OAuth2 Password Flow: JSON이 아니라 폼 형식으로 보내야 합니다.
export async function login(email, password) {
  const form = new URLSearchParams();
  form.append('username', email); // 이 백엔드는 username 칸에 이메일을 받습니다.
  form.append('password', password);
  const { data } = await api.post('/auth/token', form);
  return data; // { access_token, refresh_token, token_type }
}

export async function fetchMe() {
  const { data } = await api.get('/auth/me');
  return data; // { id, email, role }
}

export async function logout(refreshToken) {
  await api.post('/auth/logout', { refresh_token: refreshToken });
}

로그인은 S5에서 배운 대로 URLSearchParams로 폼 형식을 만들고, username 칸에 이메일을 넣습니다. axios는 URLSearchParams를 받으면 Content-Type: application/x-www-form-urlencoded를 자동으로 지정합니다.
