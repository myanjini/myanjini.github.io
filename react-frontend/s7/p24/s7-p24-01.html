도서 상세 화면

파트 목표주소의 매개변수로 데이터를 불러오는 상세 화면을 만들고, 상황별 로딩과 오류 표시를 설계할 수 있습니다.

상세 화면은 주소의 :bookId로 도서를 조회해 보여 줍니다. S9에서 대여, 수정, 삭제, 표지 업로드 버튼을 추가하므로 지금은 조회만 구현합니다.

src/pages/BookDetailPage.jsx
import { useEffect, useState } from 'react';
import { useParams } from 'react-router';
import { getBook } from '../api/books';
import { toAssetUrl } from '../api/client';
import ErrorMessage from '../components/ErrorMessage';

export default function BookDetailPage() {
  const { bookId } = useParams(); // 주소의 :bookId 값 (문자열)
  const [book, setBook] = useState(null);
  const [error, setError] = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    getBook(bookId, controller.signal)
      .then(setBook)
      .catch((err) => {
        if (err.name !== 'CanceledError') setError(err);
      });
    return () => controller.abort();
  }, [bookId]);

  if (error) return <ErrorMessage error={error} />;
  if (!book) return <p className="muted">불러오는 중입니다.</p>;

  return (
    <section className="detail">
      <div className="cover">
        {book.cover_url ? (
          <img src={toAssetUrl(book.cover_url)} alt={`${book.title} 표지`} />
        ) : (
          <div className="cover-empty">표지 없음</div>
        )}
      </div>
      <div>
        <h1>{book.title}</h1>
        <dl className="meta">
          <dt>저자</dt>
          <dd>{book.author?.name}</dd>
          <dt>가격</dt>
          <dd>{book.price.toLocaleString('ko-KR')}원</dd>
          <dt>ISBN</dt>
          <dd>{book.isbn ?? '-'}</dd>
        </dl>
      </div>
    </section>
  );
}

bookId가 바뀌면(예: 다른 도서의 링크를 누르면) 이펙트가 다시 실행되도록 의존성 배열에 넣었습니다. 없는 번호로 접근하면 백엔드의 404 응답이 ErrorMessage로 표시됩니다. 표지가 있으면 S6에서 만든 toAssetUrl로 전체 주소를 만들어 표시하고, alt 속성으로 이미지의 의미를 글로 남깁니다.

상세 화면의 라우트를 등록합니다.

src/App.jsx
import { Route, Routes } from 'react-router';
import Layout from './components/Layout';
import BookDetailPage from './pages/BookDetailPage';
import BookListPage from './pages/BookListPage';
import NotFoundPage from './pages/NotFoundPage';

export default function App() {
  return (
    <Routes>
      <Route element={<Layout />}>
        <Route index element={<BookListPage />} />
        <Route path="books/:bookId" element={<BookDetailPage />} />
        <Route path="*" element={<NotFoundPage />} />
      </Route>
    </Routes>
  );
}
