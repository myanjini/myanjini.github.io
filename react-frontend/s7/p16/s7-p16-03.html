접근 제어와 역할에 따른 화면

보호 라우트: src/auth/RequireAuth.jsx

로그인이 필요한 화면을 감싸는 컴포넌트입니다. 로그인하지 않았으면 현재 주소를 state.from에 담아 로그인 화면으로 보냅니다. adminOnly가 지정된 화면은 관리자만 볼 수 있습니다.

src/auth/RequireAuth.jsx
import { Navigate, useLocation } from 'react-router';
import { useAuth } from './useAuth';

// 로그인하지 않은 사용자를 로그인 화면으로 보냅니다.
// 화면을 가리는 장치일 뿐이며, 실제 권한 검사는 백엔드가 합니다.
export default function RequireAuth({ children, adminOnly = false }) {
  const { user, initializing, isAdmin } = useAuth();
  const location = useLocation();

  if (initializing) return <p className="muted">로그인 상태를 확인하는 중입니다.</p>;
  if (!user) return <Navigate to="/login" replace state={{ from: location.pathname }} />;
  if (adminOnly && !isAdmin) return <p className="error">관리자만 사용할 수 있는 화면입니다.</p>;
  return children;
}

레이아웃 완성: src/components/Layout.jsx

S6의 레이아웃에 로그인 상태에 따른 메뉴와 계정 영역을 추가합니다. "도서 등록"과 "내 대여"는 로그인한 사용자에게, "관제 대시보드"는 관리자에게만 보입니다. 해당 화면은 S8과 S10에서 만들며, 그 전까지는 메뉴를 누르면 404 화면이 나옵니다.

src/components/Layout.jsx
import { NavLink, Outlet, useNavigate } from 'react-router';
import { useAuth } from '../auth/useAuth';

export default function Layout() {
  const { user, isAdmin, logout } = useAuth();
  const navigate = useNavigate();

  async function handleLogout() {
    await logout();
    navigate('/');
  }

  return (
    <div className="app">
      <header className="topbar">
        <strong>도서 관리</strong>
        <nav>
          <NavLink to="/" end>도서 목록</NavLink>
          {user && <NavLink to="/books/new">도서 등록</NavLink>}
          {user && <NavLink to="/loans">내 대여</NavLink>}
          {isAdmin && <NavLink to="/dashboard">관제 대시보드</NavLink>}
        </nav>
        <div className="account">
          {user ? (
            <>
              <span>{user.email} ({user.role})</span>
              <button onClick={handleLogout}>로그아웃</button>
            </>
          ) : (
            <>
              <NavLink to="/login">로그인</NavLink>
              <NavLink to="/signup">회원가입</NavLink>
            </>
          )}
        </div>
      </header>
      <main className="content">
        <Outlet />
      </main>
    </div>
  );
}

여기서 반드시 기억해야 할 원칙이 있습니다. 메뉴를 숨기고 화면을 막는 것은 사용자 경험을 위한 장치일 뿐 보안 장치가 아닙니다. 프론트엔드 코드는 사용자의 브라우저에서 실행되므로 누구나 고칠 수 있습니다. 개발자 도구로 isAdmin을 true로 바꾸면 관리자 메뉴가 보이게 할 수도 있습니다. 그래도 안전한 이유는 백엔드가 모든 요청에서 토큰을 검증하고 권한을 다시 확인하기 때문입니다(백엔드 S5-P17의 권한 표). 프론트엔드의 권한 처리는 "어차피 거절될 동작을 미리 숨겨 혼란을 줄이는 것"이 목적입니다.

라우트 추가: src/App.jsx

로그인과 회원가입 경로를 추가합니다.

src/App.jsx
import { Route, Routes } from 'react-router';
import Layout from './components/Layout';
import BookDetailPage from './pages/BookDetailPage';
import BookListPage from './pages/BookListPage';
import LoginPage from './pages/LoginPage';
import NotFoundPage from './pages/NotFoundPage';
import SignupPage from './pages/SignupPage';

export default function App() {
  return (
    <Routes>
      <Route element={<Layout />}>
        <Route index element={<BookListPage />} />
        <Route path="books/:bookId" element={<BookDetailPage />} />
        <Route path="login" element={<LoginPage />} />
        <Route path="signup" element={<SignupPage />} />
        <Route path="*" element={<NotFoundPage />} />
      </Route>
    </Routes>
  );
}
