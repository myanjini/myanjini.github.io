저자별 보기 추가하기

파트 목표백엔드의 저자 조건으로 목록을 거르는 기능을 스스로 추가하고, 조회 화면 전체를 시나리오로 점검할 수 있습니다.

백엔드 GET /books는 author_id 쿼리로 그 저자 번호의 도서만 돌려줍니다(S1-P3-03의 API 표). 목록 화면에 저자 선택 상자를 추가합니다. 먼저 아래 요구 사항만 보고 직접 구현해 본 뒤 예시 코드와 비교합니다.

  • 저자 목록은 GET /authors로 처음 한 번만 불러옵니다. 실패해도 목록 화면은 그대로 쓸 수 있어야 합니다.
  • 선택한 저자는 다른 조건처럼 URL(?author_id=1)에 둡니다.
  • 저자를 바꾸면 첫 페이지부터 다시 봅니다.
  • "모든 저자"를 고르면 조건을 지웁니다.
src/pages/BookListPage.jsx
import { useEffect, useState } from 'react';
import { Link, useSearchParams } from 'react-router';
import { listAuthors } from '../api/authors';
import { listBooks } from '../api/books';
import ErrorMessage from '../components/ErrorMessage';
import Pagination from '../components/Pagination';

const PAGE_SIZE = 10;
const SORT_OPTIONS = [
  { value: 'id', label: '등록순' },
  { value: 'price', label: '가격 낮은 순' },
  { value: '-price', label: '가격 높은 순' },
];

export default function BookListPage() {
  // 검색 조건을 URL에 두면 새로고침, 뒤로 가기, 주소 공유에도 화면이 유지됩니다.
  const [searchParams, setSearchParams] = useSearchParams();
  const sort = searchParams.get('sort') ?? 'id';
  const skip = Number(searchParams.get('skip') ?? 0);
  const authorId = searchParams.get('author_id') ?? '';

  const q = searchParams.get('q') ?? '';
  const [keyword, setKeyword] = useState(q);

  const [authors, setAuthors] = useState([]);
  const [page, setPage] = useState(null); // { items, total, skip, limit }
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  function updateParams(changes) {
    const next = new URLSearchParams(searchParams);
    for (const [key, value] of Object.entries(changes)) {
      if (value === '' || value === null || value === undefined) next.delete(key);
      else next.set(key, String(value));
    }
    setSearchParams(next);
  }

  // 검색 버튼(또는 Enter)을 누르면 URL에 반영하고 첫 페이지부터 다시 봅니다.
  function handleSearch(event) {
    event.preventDefault();
    updateParams({ q: keyword.trim(), skip: '' });
  }

  // 저자 선택 상자에 쓸 목록은 처음 한 번만 불러옵니다.
  useEffect(() => {
    const controller = new AbortController();
    listAuthors({}, controller.signal)
      .then(setAuthors)
      .catch(() => {}); // 저자 목록이 없어도 목록 화면은 쓸 수 있으므로 조용히 넘어갑니다.
    return () => controller.abort();
  }, []);

  // URL의 검색 조건이 바뀔 때마다 목록을 다시 불러옵니다.
  useEffect(() => {
    const controller = new AbortController();
    setLoading(true);
    setError(null);
    listBooks({ q, authorId, sort, skip, limit: PAGE_SIZE }, controller.signal)
      .then(setPage)
      .catch((err) => {
        if (err.name !== 'CanceledError') setError(err);
      })
      .finally(() => {
        if (!controller.signal.aborted) setLoading(false);
      });
    // 조건이 바뀌거나 화면을 떠나면 이전 요청을 취소합니다.
    return () => controller.abort();
  }, [q, authorId, sort, skip]);

  return (
    <section>
      <h1>도서 목록</h1>
      <form className="toolbar" onSubmit={handleSearch}>
        <input
          type="search"
          placeholder="제목으로 검색"
          maxLength={50}
          value={keyword}
          onChange={(e) => setKeyword(e.target.value)}
        />
        <button type="submit">검색</button>
        <select value={authorId} onChange={(e) => updateParams({ author_id: e.target.value, skip: '' })}>
          <option value="">모든 저자</option>
          {authors.map((a) => (
            <option key={a.id} value={a.id}>
              {a.name}
            </option>
          ))}
        </select>
        <select value={sort} onChange={(e) => updateParams({ sort: e.target.value, skip: '' })}>
          {SORT_OPTIONS.map((option) => (
            <option key={option.value} value={option.value}>
              {option.label}
            </option>
          ))}
        </select>
      </form>

      <ErrorMessage error={error} />
      {loading && <p className="muted">불러오는 중입니다.</p>}

      {page && !loading && page.items.length === 0 && <p>조건에 맞는 도서가 없습니다.</p>}
      {page && page.items.length > 0 && (
        <table className="table">
          <thead>
            <tr>
              <th>번호</th>
              <th>제목</th>
              <th>저자</th>
              <th className="num">가격</th>
            </tr>
          </thead>
          <tbody>
            {page.items.map((book) => (
              <tr key={book.id}>
                <td>{book.id}</td>
                <td>
                  <Link to={`/books/${book.id}`}>{book.title}</Link>
                </td>
                <td>{book.author?.name}</td>
                <td className="num">{book.price.toLocaleString('ko-KR')}원</td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
      {page && (
        <Pagination
          total={page.total}
          skip={page.skip}
          limit={page.limit}
          onChange={(nextSkip) => updateParams({ skip: nextSkip })}
        />
      )}
    </section>
  );
}

구현에서 볼 부분은 다음과 같습니다.

  • 저자 목록 이펙트와 도서 목록 이펙트를 나눴습니다. 저자 목록은 조건과 관계없이 한 번만 필요하므로 의존성이 []이고, 도서 목록은 authorId를 포함한 조건이 바뀔 때마다 다시 불러옵니다. 하나의 이펙트에 넣으면 페이지를 넘길 때마다 저자 목록까지 다시 요청하게 됩니다.
  • 저자 목록 실패를 catch(() => {})로 조용히 넘겼습니다. 보조 기능의 실패가 핵심 기능(목록 보기)을 막지 않게 한 선택입니다. 운영 중이라면 이런 실패도 기록해 두는 것이 좋습니다(S13에서 다루는 요청 기록이 이 역할을 합니다).
  • 선택 상자의 값은 저자 이름(name)이 아니라 id입니다. 백엔드의 author_id 쿼리가 저자 번호를 받기 때문입니다. 이름은 선택 상자에 보이는 글자로만 씁니다. API 명세에 화면을 맞춘 예입니다.
  • URL에서 읽은 authorId는 "1" 같은 문자열입니다. option의 값도 문자열로 바뀌므로 선택 상자에서 그대로 맞춰지고, 백엔드는 쿼리 문자열을 정수로 바꿔 받습니다.
