헬스 체크: src/api/health.js

헬스 체크는 Fetch API로 작성합니다. 데이터베이스 장애 시 백엔드는 503과 함께 {"status": "error", "database": "unreachable"}를 보내는데, 관제 화면에는 이 본문 자체가 중요한 정보입니다. S5-P10-02에서 배운 대로 fetch는 503도 정상 응답으로 돌려주므로 본문을 그대로 읽을 수 있습니다. 응답을 받지 못한 경우(서버 중단, 네트워크 단절)는 catch에서 unreachable로 기록합니다. 응답 시간도 함께 재어 S10의 대시보드에서 사용합니다.

src/api/health.js
import { API_BASE_URL } from './client';

// 헬스 체크는 Fetch API로 직접 호출합니다.
// 503도 의미 있는 응답이므로 상태 코드와 본문을 함께 돌려줍니다.
export async function checkHealth(signal) {
  const startedAt = performance.now();
  try {
    const response = await fetch(`${API_BASE_URL}/health`, { signal, cache: 'no-store' });
    const body = await response.json().catch(() => null);
    return {
      reachable: true,
      httpStatus: response.status,
      status: body?.status ?? 'unknown',
      database: body?.database ?? 'unknown',
      durationMs: Math.round(performance.now() - startedAt),
      checkedAt: Date.now(),
    };
  } catch (error) {
    if (error.name === 'AbortError') throw error;
    return {
      reachable: false,
      httpStatus: 0,
      status: 'unreachable',
      database: 'unknown',
      durationMs: null,
      checkedAt: Date.now(),
    };
  }
}
