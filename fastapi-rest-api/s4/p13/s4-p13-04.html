저자 API와 관계 조회

저자 테이블과 저자 API를 만들고, 도서를 저자와 연결합니다. 마지막에는 관계 조회에서 흔히 생기는 N+1 문제를 직접 확인하고 고칩니다.

실습 목표1:N 관계를 정의하고, 저자 API와 저자 정보가 중첩된 도서 응답을 구현할 수 있습니다.

  1. 서버를 끄고 데이터베이스 파일 지우기

    이번에는 books 테이블의 열이 바뀝니다(author가 빠지고 author_id가 생김). 그런데 create_all은 없는 테이블만 만들고 있는 테이블은 고치지 못합니다. 그래서 개발용 데이터를 버리고 books.db를 지운 뒤 새로 만듭니다.

    지금만 쓰는 방법입니다

    실제 서비스에서는 데이터베이스를 지울 수 없습니다. 데이터를 지키면서 테이블을 바꾸는 방법은 바로 다음 파트(S4-P14)에서 배웁니다.

    from pydantic import BaseModel, ConfigDict, Field


class AuthorCreate(BaseModel):
    name: str = Field(min_length=1, max_length=50, examples=["김하늘"])


class AuthorRead(BaseModel):
    model_config = ConfigDict(from_attributes=True)

    id: int
    name: str


class BookCreate(BaseModel):
    """등록에 쓰는 입력 모델"""
    title: str = Field(min_length=1, max_length=100, examples=["FastAPI 실전"])
    author_id: int = Field(gt=0, examples=[1])
    price: int = Field(ge=0, le=1_000_000, examples=[27000])
    isbn: str | None = Field(default=None, pattern=r"^\d{13}$", examples=["9791100000035"])


class BookUpdate(BaseModel):
    """일부 수정(PATCH)에 쓰는 입력 모델. 모든 항목이 선택입니다."""
    title: str | None = Field(default=None, min_length=1, max_length=100)
    author_id: int | None = Field(default=None, gt=0)
    price: int | None = Field(default=None, ge=0, le=1_000_000)
    isbn: str | None = Field(default=None, pattern=r"^\d{13}$")


class BookRead(BaseModel):
    """응답에 쓰는 출력 모델. 저자는 이름까지 담아 중첩해서 보냅니다."""

    model_config = ConfigDict(from_attributes=True)

    id: int
    title: str
    price: int
    isbn: str | None = None
    author: AuthorRead


class BookPage(BaseModel):
    """목록 응답. 현재 페이지의 도서와 전체 개수를 함께 돌려줍니다."""
    items: list[BookRead]
    total: int
    skip: int
    limit: int

  3. 저자 라우터 만들기

    app/routers/authors.py
    from fastapi import APIRouter, status
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.dependencies import PageDep, SessionDep
from app.errors import AppError
from app.models import Author, Book
from app.schemas import AuthorCreate, AuthorRead, BookRead

router = APIRouter(prefix="/authors", tags=["저자"])


def find_author(db: Session, author_id: int) -> Author:
    author = db.get(Author, author_id)
    if author is None:
        raise AppError(404, "AUTHOR_NOT_FOUND", f"{author_id}번 저자를 찾을 수 없습니다.")
    return author


@router.post("", response_model=AuthorRead, status_code=status.HTTP_201_CREATED, summary="저자 등록",
             responses={409: {"description": "이미 등록된 저자 이름"}})
def create_author(db: SessionDep, author_in: AuthorCreate):
    if db.scalar(select(Author).where(Author.name == author_in.name)) is not None:
        raise AppError(409, "DUPLICATE_AUTHOR", "이미 등록된 저자입니다.")
    author = Author(name=author_in.name)
    db.add(author)
    db.commit()
    db.refresh(author)
    return author


@router.get("", response_model=list[AuthorRead], summary="저자 목록 조회")
def list_authors(db: SessionDep, page: PageDep):
    stmt = select(Author).order_by(Author.id).offset(page.skip).limit(page.limit)
    return db.scalars(stmt).all()


@router.get("/{author_id}", response_model=AuthorRead, summary="저자 상세 조회",
            responses={404: {"description": "저자를 찾을 수 없음"}})
def get_author(db: SessionDep, author_id: int):
    return find_author(db, author_id)


@router.get("/{author_id}/books", response_model=list[BookRead], summary="저자별 도서 목록")
def list_author_books(db: SessionDep, author_id: int):
    author = find_author(db, author_id)
    return author.books


@router.delete("/{author_id}", status_code=status.HTTP_204_NO_CONTENT, summary="저자 삭제",
               responses={404: {"description": "저자를 찾을 수 없음"}, 409: {"description": "도서가 남아 있는 저자"}})
def delete_author(db: SessionDep, author_id: int):
    author = find_author(db, author_id)
    if author.books:
        raise AppError(409, "AUTHOR_HAS_BOOKS", "이 저자의 도서가 남아 있어 삭제할 수 없습니다.")
    db.delete(author)
    db.commit()

    list_author_books는 관계를 따라 author.books를 그대로 돌려줍니다. 저자를 삭제할 때 도서가 남아 있으면 409로 막아, 주인 없는 도서가 생기지 않게 합니다.

  7. N+1 문제 확인

    app/database.py의 엔진에 echo=True를 잠시 붙이면 실행되는 SQL이 터미널에 모두 찍힙니다.

    app/database.py
    engine = create_engine(settings.database_url, connect_args=connect_args, echo=True)

    GET /books를 호출하고 터미널을 봅니다. 도서 목록을 가져오는 SELECT 한 번 뒤에, 저자를 가져오는 SELECT ... FROM authors가 저자 수만큼 이어서 찍힙니다. 응답을 만들면서 도서마다 book.author를 처음 읽을 때 그 저자를 따로 가져오기 때문입니다. 목록 1번에 저자 N번이 더해진다고 해서 N+1 문제라고 부릅니다. 저자가 100명이면 쿼리가 100번 넘게 실행됩니다.

  8. selectinload로 고치기

    app/routers/books.py
    from sqlalchemy.orm import Session, selectinload

    stmt = (
        select(Book)
        .options(selectinload(Book.author))  # 저자를 한 번에 함께 읽어 옵니다.
        .where(*conditions)
        ...
    )

    다시 GET /books를 호출하면, 도서를 가져온 뒤 저자를 WHERE authors.id IN (1, 2, 3)처럼 한 번에 가져옵니다. 저자가 몇 명이든 쿼리 수가 늘지 않습니다. 확인했으면 echo=True를 지웁니다.

전체 코드 펼치기
app/routers/books.py
from typing import Literal

from fastapi import APIRouter, Query, status
from sqlalchemy import func, select
from sqlalchemy.orm import Session, selectinload

from app.dependencies import PageDep, SessionDep
from app.errors import AppError
from app.models import Author, Book
from app.schemas import BookCreate, BookPage, BookRead, BookUpdate

router = APIRouter(prefix="/books", tags=["도서"])


def find_book(db: Session, book_id: int) -> Book:
    book = db.get(Book, book_id)
    if book is None:
        raise AppError(404, "BOOK_NOT_FOUND", f"{book_id}번 도서를 찾을 수 없습니다.")
    return book


def check_isbn_unique(db: Session, isbn: str | None, exclude_id: int | None = None) -> None:
    if isbn is None:
        return
    found = db.scalar(select(Book).where(Book.isbn == isbn))
    if found is not None and found.id != exclude_id:
        raise AppError(409, "DUPLICATE_ISBN", "이미 등록된 ISBN입니다.")


def check_author_exists(db: Session, author_id: int | None) -> None:
    """요청에 적힌 저자 번호가 실제로 있는지 확인합니다."""
    if author_id is None:
        return
    if db.get(Author, author_id) is None:
        raise AppError(422, "INVALID_AUTHOR", f"{author_id}번 저자가 없습니다.")


# 정렬 기준으로 받을 수 있는 값과 실제 정렬 방법을 짝지어 둡니다.
SORT_OPTIONS = {
    "id": Book.id,
    "price": Book.price,
    "-price": Book.price.desc(),
}


@router.get("", response_model=BookPage, summary="도서 목록 조회")
def list_books(
    db: SessionDep,
    page: PageDep,
    q: str | None = Query(default=None, max_length=50, description="제목에 들어간 글자로 검색"),
    author_id: int | None = Query(default=None, description="저자 번호"),
    sort: Literal["id", "price", "-price"] = Query(default="id", description="정렬 기준"),
):
    conditions = []
    if q is not None:
        conditions.append(Book.title.contains(q))
    if author_id is not None:
        conditions.append(Book.author_id == author_id)

    total = db.scalar(select(func.count()).select_from(Book).where(*conditions))
    stmt = (
        select(Book)
        .options(selectinload(Book.author))  # 저자를 한 번에 함께 읽어 옵니다.
        .where(*conditions)
        .order_by(SORT_OPTIONS[sort], Book.id)
        .offset(page.skip)
        .limit(page.limit)
    )
    items = db.scalars(stmt).all()
    return {"items": items, "total": total, "skip": page.skip, "limit": page.limit}


@router.get("/{book_id}", response_model=BookRead, summary="도서 상세 조회",
            responses={404: {"description": "도서를 찾을 수 없음"}})
def get_book(db: SessionDep, book_id: int):
    return find_book(db, book_id)


@router.post("", response_model=BookRead, status_code=status.HTTP_201_CREATED, summary="도서 등록",
             responses={409: {"description": "이미 등록된 ISBN"}})
def create_book(db: SessionDep, book_in: BookCreate):
    check_isbn_unique(db, book_in.isbn)
    check_author_exists(db, book_in.author_id)
    book = Book(**book_in.model_dump())
    db.add(book)      # 저장할 목록에 올립니다.
    db.commit()       # 데이터베이스에 실제로 저장합니다.
    db.refresh(book)  # 데이터베이스가 정한 id를 다시 읽어 옵니다.
    return book


@router.patch("/{book_id}", response_model=BookRead, summary="도서 일부 수정",
              responses={404: {"description": "도서를 찾을 수 없음"}, 409: {"description": "이미 등록된 ISBN"}})
def update_book(db: SessionDep, book_id: int, book_in: BookUpdate):
    book = find_book(db, book_id)
    changes = book_in.model_dump(exclude_unset=True, exclude_none=True)
    check_isbn_unique(db, changes.get("isbn"), exclude_id=book_id)
    check_author_exists(db, changes.get("author_id"))
    for field, value in changes.items():
        setattr(book, field, value)  # book.price = 19000 과 같은 뜻입니다.
    db.commit()
    db.refresh(book)
    return book


@router.delete("/{book_id}", status_code=status.HTTP_204_NO_CONTENT, summary="도서 삭제",
               responses={404: {"description": "도서를 찾을 수 없음"}})
def delete_book(db: SessionDep, book_id: int):
    book = find_book(db, book_id)
    db.delete(book)
    db.commit()
