서비스 계층 적용

도서 API를 세 계층으로 나누고 재고 기능을 추가합니다. 마지막에는 일부러 오류를 내서 트랜잭션이 정말 "전부 아니면 전무"로 동작하는지 눈으로 확인합니다.

실습 목표라우터, 서비스, 저장소를 분리하고, 도서 등록과 재고 기록을 하나의 트랜잭션으로 처리할 수 있습니다.

  1. 모델에 재고와 재고 기록 추가

    서버를 끄고 app/models.py를 고칩니다. 가져오기에 func를 더하고, Book의 owner_id 아래에 두 줄을, 파일 맨 아래에 StockLog를 추가합니다.

    app/models.py
    from sqlalchemy import ForeignKey, String, func
    app/models.py
        # 재고 수량. 기존 도서는 0권으로 시작합니다.
    stock: Mapped[int] = mapped_column(default=0, server_default="0")
    # 이 도서의 재고 기록. 도서를 지우면 기록도 함께 지웁니다.
    stock_logs: Mapped[list["StockLog"]] = relationship(cascade="all, delete-orphan")
    app/models.py
    class StockLog(Base):
    """재고가 언제, 얼마나, 왜 바뀌었는지 남기는 기록입니다."""

    __tablename__ = "stock_logs"

    id: Mapped[int] = mapped_column(primary_key=True)
    book_id: Mapped[int] = mapped_column(ForeignKey("books.id"))
    change: Mapped[int]  # 늘어나면 양수, 줄어들면 음수
    reason: Mapped[str] = mapped_column(String(20))
    created_at: Mapped[datetime] = mapped_column(server_default=func.now())  # 데이터베이스가 현재 시각을 넣습니다.

    stock_logs의 cascade="all, delete-orphan"은 도서를 지울 때 그 도서의 재고 기록도 함께 지우라는 뜻입니다. 기록이 남아 있으면 지워진 도서를 가리키는 행이 생기기 때문입니다.

    app/services 폴더를 만들고 빈 __init__.py와 books.py를 넣습니다. 검사 함수들은 라우터에 있던 코드에서 쿼리 부분만 저장소 호출로 바꾼 것입니다.

    app/services/books.py 전체 펼치기
    app/services/books.py
    """도서 서비스: 규칙 검사와 트랜잭션(commit)을 맡습니다. HTTP 요청과 응답은 모릅니다."""
from sqlalchemy.orm import Session

from app.errors import AppError
from app.models import Book, User
from app.repositories import authors as author_repo
from app.repositories import books as book_repo
from app.schemas import BookCreate, BookUpdate, StockChange


def find_book(db: Session, book_id: int) -> Book:
    book = book_repo.get(db, book_id)
    if book is None:
        raise AppError(404, "BOOK_NOT_FOUND", f"{book_id}번 도서를 찾을 수 없습니다.")
    return book


def check_isbn_unique(db: Session, isbn: str | None, exclude_id: int | None = None) -> None:
    if isbn is None:
        return
    found = book_repo.get_by_isbn(db, isbn)
    if found is not None and found.id != exclude_id:
        raise AppError(409, "DUPLICATE_ISBN", "이미 등록된 ISBN입니다.")


def check_author_exists(db: Session, author_id: int | None) -> None:
    if author_id is None:
        return
    if author_repo.get(db, author_id) is None:
        raise AppError(422, "INVALID_AUTHOR", f"{author_id}번 저자가 없습니다.")


def check_can_modify(book: Book, user: User) -> None:
    """도서를 등록한 사람이나 관리자만 수정, 삭제할 수 있습니다."""
    if user.role == "admin" or book.owner_id == user.id:
        return
    raise AppError(403, "FORBIDDEN", "이 도서를 수정하거나 삭제할 권한이 없습니다.")


def change_stock(db: Session, book: Book, change: int, reason: str) -> None:
    """재고를 바꾸고 기록을 남깁니다. 확정(commit)은 이 함수를 부른 쪽에서 합니다."""
    if book.stock + change < 0:
        raise AppError(409, "INSUFFICIENT_STOCK", f"재고가 부족합니다. 현재 재고는 {book.stock}권입니다.")
    book.stock += change
    book_repo.add_stock_log(db, book, change, reason)


def create_book(db: Session, user: User, book_in: BookCreate) -> Book:
    check_isbn_unique(db, book_in.isbn)
    check_author_exists(db, book_in.author_id)
    book = book_repo.add(db, Book(**book_in.model_dump(exclude={"stock"}), owner_id=user.id))
    if book_in.stock > 0:
        change_stock(db, book, book_in.stock, "입고")
    db.commit()  # 도서 등록과 재고 기록을 한 번에 확정합니다.
    db.refresh(book)
    return book


def list_books(db: Session, q: str | None, author_id: int | None, sort: str,
               skip: int, limit: int) -> dict:
    items, total = book_repo.search(db, q, author_id, sort, skip, limit)
    return {"items": items, "total": total, "skip": skip, "limit": limit}


def update_book(db: Session, user: User, book_id: int, book_in: BookUpdate) -> Book:
    book = find_book(db, book_id)
    check_can_modify(book, user)
    changes = book_in.model_dump(exclude_unset=True, exclude_none=True)
    check_isbn_unique(db, changes.get("isbn"), exclude_id=book_id)
    check_author_exists(db, changes.get("author_id"))
    for field, value in changes.items():
        setattr(book, field, value)
    db.commit()
    db.refresh(book)
    return book


def delete_book(db: Session, user: User, book_id: int) -> None:
    book = find_book(db, book_id)
    check_can_modify(book, user)
    book_repo.delete(db, book)  # 재고 기록도 함께 지워집니다(cascade).
    db.commit()


def adjust_stock(db: Session, user: User, book_id: int, data: StockChange) -> Book:
    book = find_book(db, book_id)
    check_can_modify(book, user)
    change_stock(db, book, data.change, data.reason)
    db.commit()
    db.refresh(book)
    return book

  7. 라우터를 얇게 바꾸기

    app/routers/books.py를 아래 전체 코드로 바꿉니다. 경로 함수마다 서비스 함수를 한 번 부르는 것이 전부입니다. 재고 변경 API(POST /books/{book_id}/stock)가 새로 생겼습니다.

    app/routers/books.py 전체 펼치기
    app/routers/books.py
    from typing import Literal

from fastapi import APIRouter, Query, status

from app.dependencies import CurrentUserDep, PageDep, SessionDep
from app.schemas import BookCreate, BookPage, BookRead, BookUpdate, StockChange
from app.services import books as book_service

router = APIRouter(prefix="/books", tags=["도서"])

FORBIDDEN_OR_NOT_FOUND = {401: {"description": "로그인 필요"}, 403: {"description": "권한 없음"},
                          404: {"description": "도서를 찾을 수 없음"}}


@router.get("", response_model=BookPage, summary="도서 목록 조회")
def list_books(
    db: SessionDep,
    page: PageDep,
    q: str | None = Query(default=None, max_length=50, description="제목에 들어간 글자로 검색"),
    author_id: int | None = Query(default=None, description="저자 번호"),
    sort: Literal["id", "price", "-price"] = Query(default="id", description="정렬 기준"),
):
    return book_service.list_books(db, q, author_id, sort, page.skip, page.limit)


@router.get("/{book_id}", response_model=BookRead, summary="도서 상세 조회",
            responses={404: {"description": "도서를 찾을 수 없음"}})
def get_book(db: SessionDep, book_id: int):
    return book_service.find_book(db, book_id)


@router.post("", response_model=BookRead, status_code=status.HTTP_201_CREATED, summary="도서 등록",
             responses={401: {"description": "로그인 필요"}, 409: {"description": "이미 등록된 ISBN"}})
def create_book(db: SessionDep, user: CurrentUserDep, book_in: BookCreate):
    return book_service.create_book(db, user, book_in)


@router.patch("/{book_id}", response_model=BookRead, summary="도서 일부 수정",
              responses={**FORBIDDEN_OR_NOT_FOUND, 409: {"description": "이미 등록된 ISBN"}})
def update_book(db: SessionDep, user: CurrentUserDep, book_id: int, book_in: BookUpdate):
    return book_service.update_book(db, user, book_id, book_in)


@router.delete("/{book_id}", status_code=status.HTTP_204_NO_CONTENT, summary="도서 삭제",
               responses=FORBIDDEN_OR_NOT_FOUND)
def delete_book(db: SessionDep, user: CurrentUserDep, book_id: int):
    book_service.delete_book(db, user, book_id)


@router.post("/{book_id}/stock", response_model=BookRead, summary="재고 변경",
             responses={**FORBIDDEN_OR_NOT_FOUND, 409: {"description": "재고 부족"}})
def adjust_stock(db: SessionDep, user: CurrentUserDep, book_id: int, data: StockChange):
    return book_service.adjust_stock(db, user, book_id, data)

  8. 기존 기능과 재고 확인

    서버를 실행하고 Authorize로 user1이 로그인합니다. 먼저 기존 기능이 그대로인지 확인합니다. 계층을 나누는 작업은 동작을 바꾸지 않고 구조만 바꾸는 것이므로, 결과가 달라지면 옮기다가 실수한 것입니다.

    요청기대하는 결과내 결과
    GET /books200. 기존 도서는 "stock": 0
    POST /books {"title": "트랜잭션 입문", "author_id": 1, "price": 21000, "stock": 3}201, "stock": 3. 새 번호를 기억합니다(이 교안에서는 6).
    POST /books/6/stock {"change": -5, "reason": "분실"}409, INSUFFICIENT_STOCK, 현재 재고는 3권입니다.
    POST /books/6/stock {"change": 2, "reason": "추가 입고"}200, "stock": 5
    PATCH /books/6 {"price": 22000}200

    재고 기록이 쌓였는지 확인합니다. (6, 3, '입고'), (6, 2, '추가 입고') 두 줄이 나와야 합니다. 409로 실패한 요청은 기록을 남기지 않았습니다.

  9. 일부러 실패시켜 보기

    app/services/books.py의 create_book에서 book_repo.add 바로 아래에 한 줄을 넣고 서버를 다시 시작합니다.

    app/services/books.py
        book = book_repo.add(db, Book(**book_in.model_dump(exclude={"stock"}), owner_id=user.id))
    raise RuntimeError("실습용으로 일부러 낸 오류")  # 확인 후 반드시 지웁니다.

    POST /books로 {"title": "실패 실험", "author_id": 1, "price": 1000, "stock": 2}를 보내면 500 오류가 납니다. GET /books?q=실패로 찾아보면 total이 0입니다. flush로 INSERT까지 실행했지만 확정하지 않았으므로 rollback으로 사라졌습니다.

  10. 중간에 commit하면 어떻게 되는지 보기

    이번에는 오류 줄 위에 db.commit()을 넣습니다.

    app/services/books.py
        book = book_repo.add(db, Book(**book_in.model_dump(exclude={"stock"}), owner_id=user.id))
    db.commit()  # 잘못된 예: 중간에 확정
    raise RuntimeError("실습용으로 일부러 낸 오류")  # 확인 후 반드시 지웁니다.

    같은 요청을 보내면 역시 500이지만, GET /books?q=실패에 "실패 실험" 도서가 "stock": 0으로 남아 있습니다. 재고 3권도, 입고 기록도 없는 어중간한 데이터입니다. 트랜잭션 중간에 확정하면 이런 일이 생깁니다. 확인했으면 넣었던 두 줄을 반드시 지우고, 남은 "실패 실험" 도서는 DELETE로 지웁니다.
