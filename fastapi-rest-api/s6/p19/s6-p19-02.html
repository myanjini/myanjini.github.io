저장소

app/repositories/books.py
"""도서 저장소: 데이터베이스에서 도서를 읽고 쓰는 코드만 둡니다. commit은 하지 않습니다."""
from sqlalchemy import func, select
from sqlalchemy.orm import Session, selectinload

from app.models import Book, StockLog

# 정렬 기준으로 받을 수 있는 값과 실제 정렬 방법을 짝지어 둡니다.
SORT_OPTIONS = {
    "id": Book.id,
    "price": Book.price,
    "-price": Book.price.desc(),
}


def get(db: Session, book_id: int) -> Book | None:
    return db.get(Book, book_id)


def get_by_isbn(db: Session, isbn: str) -> Book | None:
    return db.scalar(select(Book).where(Book.isbn == isbn))


def search(db: Session, q: str | None, author_id: int | None, sort: str,
           skip: int, limit: int) -> tuple[list[Book], int]:
    """조건에 맞는 도서 한 페이지와 전체 개수를 돌려줍니다."""
    conditions = []
    if q is not None:
        conditions.append(Book.title.contains(q))
    if author_id is not None:
        conditions.append(Book.author_id == author_id)

    total = db.scalar(select(func.count()).select_from(Book).where(*conditions))
    stmt = (
        select(Book)
        .options(selectinload(Book.author))
        .where(*conditions)
        .order_by(SORT_OPTIONS[sort], Book.id)
        .offset(skip)
        .limit(limit)
    )
    return list(db.scalars(stmt).all()), total


def add(db: Session, book: Book) -> Book:
    db.add(book)
    db.flush()  # INSERT를 먼저 실행해 book.id를 받습니다. 확정(commit)은 아직 하지 않습니다.
    return book


def delete(db: Session, book: Book) -> None:
    db.delete(book)


def add_stock_log(db: Session, book: Book, change: int, reason: str) -> None:
    db.add(StockLog(book_id=book.id, change=change, reason=reason))

add에서 flush를 부르는 이유는 등록한 도서의 번호가 바로 필요하기 때문입니다. 재고 기록에 book_id를 넣어야 하는데, 번호는 데이터베이스가 INSERT를 실행해야 정해집니다. flush는 INSERT를 실행하지만 확정하지 않으므로, 뒤에서 실패하면 이 INSERT도 함께 취소됩니다.

서비스: 재고 규칙과 트랜잭션

app/services/books.py
def change_stock(db: Session, book: Book, change: int, reason: str) -> None:
    """재고를 바꾸고 기록을 남깁니다. 확정(commit)은 이 함수를 부른 쪽에서 합니다."""
    if book.stock + change < 0:
        raise AppError(409, "INSUFFICIENT_STOCK", f"재고가 부족합니다. 현재 재고는 {book.stock}권입니다.")
    book.stock += change
    book_repo.add_stock_log(db, book, change, reason)
app/services/books.py
def create_book(db: Session, user: User, book_in: BookCreate) -> Book:
    check_isbn_unique(db, book_in.isbn)
    check_author_exists(db, book_in.author_id)
    book = book_repo.add(db, Book(**book_in.model_dump(exclude={"stock"}), owner_id=user.id))
    if book_in.stock > 0:
        change_stock(db, book, book_in.stock, "입고")
    db.commit()  # 도서 등록과 재고 기록을 한 번에 확정합니다.
    db.refresh(book)
    return book

change_stock은 재고를 바꾸고 기록을 남기지만 commit하지 않습니다. 그래서 도서 등록 안에서도, 나중에 만들 대여 기능 안에서도 부를 수 있고, 부른 쪽의 트랜잭션에 자연스럽게 포함됩니다. create_book은 도서 INSERT, 재고 변경, 재고 기록 INSERT를 모두 마친 뒤 한 번만 commit합니다.

오류가 나면 되돌리기

서비스 중간에 예외가 나면 commit까지 가지 못합니다. 이때 확정하지 않은 변경을 확실히 취소하도록 세션을 여는 get_db에서 rollback합니다.

app/database.py
def get_db():
    """요청 하나에 세션 하나를 열어 주고, 요청이 끝나면 닫습니다."""
    db = SessionLocal()
    try:
        yield db
    except Exception:
        db.rollback()  # 요청 처리 중 오류가 나면 확정하지 않은 변경을 모두 되돌립니다.
        raise
    finally:
        db.close()

경로 함수에서 예외가 나면 FastAPI가 그 예외를 yield 자리로 다시 던져 줍니다. except에서 되돌린 뒤 raise로 예외를 그대로 넘겨야 원래의 오류 응답(409, 500 등)이 나갑니다. 사실 close()만 해도 확정하지 않은 변경은 버려지지만, 의도를 코드로 분명히 드러내려고 rollback을 적습니다.
