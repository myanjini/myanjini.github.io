핵심 CRUD 구현

저장소, 서비스, 라우터 순서로 주 자원의 CRUD를 만들고 Swagger UI로 확인합니다. 권한 검사는 다음 파트에서 넣습니다.

실습 목표명세대로 주 자원의 등록, 목록, 상세, 수정, 삭제 API를 구현하고 Swagger UI로 확인할 수 있습니다.

  1. 저장소

    app/repositories/studies.py를 새로 만듭니다. 아래는 완성본이며, 지금은 위쪽의 get부터 delete까지만 쓰면 됩니다. 나머지(참가 관련 함수)는 S17-P60-01에서 씁니다.

    app/repositories/studies.py
    """모임 저장소: 데이터베이스에서 모임과 참가 기록을 읽고 쓰는 코드만 둡니다. commit은 하지 않습니다."""
from sqlalchemy import func, select, update
from sqlalchemy.orm import Session, selectinload

from app.models import Membership, Study


def get(db: Session, study_id: int) -> Study | None:
    return db.get(Study, study_id)


def search(db: Session, keyword: str | None, offset: int, limit: int) -> tuple[list[Study], int]:
    """제목에 keyword가 들어간 모임 한 페이지와 전체 개수를 돌려줍니다."""
    conditions = [Study.title.contains(keyword)] if keyword else []
    total = db.scalar(select(func.count()).select_from(Study).where(*conditions))
    stmt = select(Study).where(*conditions).order_by(Study.id.desc()).offset(offset).limit(limit)
    return list(db.scalars(stmt).all()), total


def add(db: Session, study: Study) -> Study:
    db.add(study)
    db.flush()  # INSERT를 먼저 실행해 study.id를 받습니다.
    return study


def delete(db: Session, study: Study) -> None:
    db.delete(study)


def find_membership(db: Session, study_id: int, user_id: int) -> Membership | None:
    return db.scalar(select(Membership).where(Membership.study_id == study_id, Membership.user_id == user_id))


def add_membership(db: Session, study_id: int, user_id: int) -> None:
    db.add(Membership(study_id=study_id, user_id=user_id))
    db.flush()  # 같은 사람이 두 번 참가하면 여기서 IntegrityError가 납니다.


def increase_member_count(db: Session, study_id: int) -> bool:
    """모집 중이고 자리가 남았을 때만 인원을 1 늘립니다. 늘렸으면 True를 돌려줍니다."""
    stmt = (
        update(Study)
        .where(Study.id == study_id, Study.is_open.is_(True), Study.member_count < Study.capacity)
        .values(member_count=Study.member_count + 1)  # 지금 값에 더하는 계산을 데이터베이스가 합니다.
        .execution_options(synchronize_session=False)
    )
    return db.execute(stmt).rowcount == 1


def decrease_member_count(db: Session, study_id: int) -> None:
    stmt = (update(Study).where(Study.id == study_id)
            .values(member_count=Study.member_count - 1).execution_options(synchronize_session=False))
    db.execute(stmt)


def list_members(db: Session, study_id: int) -> list[Membership]:
    stmt = (select(Membership).options(selectinload(Membership.user))
            .where(Membership.study_id == study_id).order_by(Membership.id))
    return list(db.scalars(stmt).all())


def list_joined(db: Session, user_id: int) -> list[Study]:
    """이 사용자가 참가한(개설한 모임 포함) 모임 목록"""
    stmt = (select(Study).join(Membership).where(Membership.user_id == user_id)
            .order_by(Membership.id.desc()))
    return list(db.scalars(stmt).all())

  2. 서비스

    app/services/studies.py를 새로 만듭니다. 이 단계의 서비스에는 아직 권한 검사가 없습니다. 주석으로 자리만 표시해 둡니다.

    app/services/studies.py(이 단계)
    """모임 서비스: 규칙과 권한 검사를 둡니다."""
import logging

from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from app.errors import AppError
from app.models import Membership, Study, User
from app.repositories import studies as study_repo
from app.schemas import StudyCreate, StudyUpdate

PAGE_SIZE = 10  # 목록 한 페이지에 보여 줄 모임 수
logger = logging.getLogger(__name__)


def find_study(db: Session, study_id: int) -> Study:
    study = study_repo.get(db, study_id)
    if study is None:
        raise AppError(404, "STUDY_NOT_FOUND", f"{study_id}번 모임을 찾을 수 없습니다.")
    return study


def create_study(db: Session, user: User, study_in: StudyCreate) -> Study:
    study = study_repo.add(db, Study(**study_in.model_dump(), owner_id=user.id))
    db.add(Membership(study_id=study.id, user_id=user.id))  # 개설자는 첫 참가자입니다.
    db.commit()
    db.refresh(study)
    logger.info("모임 등록", extra={"user_id": user.id, "study_id": study.id})
    return study


def list_studies(db: Session, keyword: str | None, page: int) -> dict:
    items, total = study_repo.search(db, keyword, (page - 1) * PAGE_SIZE, PAGE_SIZE)
    return {"items": items, "total": total, "page": page, "size": PAGE_SIZE}


def update_study(db: Session, user: User, study_id: int, study_in: StudyUpdate) -> Study:
    study = find_study(db, study_id)
    # 권한 검사는 S17-P59-02에서 넣습니다.
    changes = study_in.model_dump(exclude_unset=True)  # 보낸 항목만 바꿉니다.
    if changes.get("capacity") is not None and changes["capacity"] < study.member_count:
        raise AppError(409, "CAPACITY_TOO_SMALL", f"정원을 지금 인원({study.member_count}명)보다 적게 줄일 수 없습니다.")
    for key, value in changes.items():
        if value is not None:  # null로 보낸 항목은 바꾸지 않습니다.
            setattr(study, key, value)
    db.commit()
    db.refresh(study)
    return study


def delete_study(db: Session, user: User, study_id: int) -> None:
    study = find_study(db, study_id)
    # 권한 검사는 S17-P59-02에서 넣습니다.
    study_repo.delete(db, study)
    db.commit()
    logger.info("모임 삭제", extra={"user_id": user.id, "study_id": study_id})

    create_study에서 owner_id는 요청 본문이 아니라 로그인한 사용자(user.id)에서 가져옵니다. update_study는 exclude_unset=True로 보낸 항목만 바꾸고, 정원을 지금 인원보다 적게 줄이려 하면 409로 막습니다. 이 규칙은 입력 검증(2~50)만으로는 막을 수 없는, 지금 상태에 따른 규칙입니다.

  3. 라우터와 등록

    app/routers/studies.py를 만들고, app/main.py에 등록합니다.

    app/routers/studies.py(이 단계)
    from fastapi import APIRouter, Query, status

from app.dependencies import CurrentUserDep, SessionDep
from app.schemas import MemberRead, StudyCreate, StudyPage, StudyRead, StudyUpdate
from app.services import studies as study_service

router = APIRouter(prefix="/studies", tags=["모임"])

NEED_OWNER = {401: {"description": "로그인 필요"}, 403: {"description": "개설자가 아님"},
              404: {"description": "모임을 찾을 수 없음"}}


@router.get("", response_model=StudyPage, summary="모임 목록 조회")
def list_studies(
    db: SessionDep,
    keyword: str | None = Query(default=None, max_length=50, description="제목에 들어간 글자로 검색"),
    page: int = Query(default=1, ge=1, le=1000, description="페이지 번호(1부터)"),
):
    return study_service.list_studies(db, keyword, page)


@router.get("/{study_id}", response_model=StudyRead, summary="모임 상세 조회",
            responses={404: {"description": "모임을 찾을 수 없음"}})
def get_study(db: SessionDep, study_id: int):
    return study_service.find_study(db, study_id)


@router.post("", response_model=StudyRead, status_code=status.HTTP_201_CREATED, summary="모임 등록",
             responses={401: {"description": "로그인 필요"}})
def create_study(db: SessionDep, user: CurrentUserDep, study_in: StudyCreate):
    return study_service.create_study(db, user, study_in)


@router.patch("/{study_id}", response_model=StudyRead, summary="모임 일부 수정",
              responses={**NEED_OWNER, 409: {"description": "정원을 지금 인원보다 적게 줄임"}})
def update_study(db: SessionDep, user: CurrentUserDep, study_id: int, study_in: StudyUpdate):
    return study_service.update_study(db, user, study_id, study_in)


@router.delete("/{study_id}", status_code=status.HTTP_204_NO_CONTENT, summary="모임 삭제",
               responses=NEED_OWNER)
def delete_study(db: SessionDep, user: CurrentUserDep, study_id: int):
    study_service.delete_study(db, user, study_id)
