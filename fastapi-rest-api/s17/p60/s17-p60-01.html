예시: 참가 신청과 정원

예시 프로젝트의 핵심은 "정원을 넘지 않는 참가 신청"입니다. 저장소에서 미리 만들어 둔 increase_member_count가 경쟁 조건을 막는 핵심입니다. 인원 확인과 증가를 데이터베이스가 한 문장으로 처리하므로, 마지막 자리에 동시에 여러 요청이 와도 한 요청만 성공합니다.

app/repositories/studies.py(일부)
def increase_member_count(db: Session, study_id: int) -> bool:
    """모집 중이고 자리가 남았을 때만 인원을 1 늘립니다. 늘렸으면 True를 돌려줍니다."""
    stmt = (
        update(Study)
        .where(Study.id == study_id, Study.is_open.is_(True), Study.member_count < Study.capacity)
        .values(member_count=Study.member_count + 1)  # 지금 값에 더하는 계산을 데이터베이스가 합니다.
        .execution_options(synchronize_session=False)
    )
    return db.execute(stmt).rowcount == 1

서비스에는 참가, 나가기, 명단, 내 모임을 더합니다. 같은 사람의 동시 요청은 모델의 유일 제약이 막고, 그때 나는 IntegrityError를 409로 바꿉니다.

app/services/studies.py에 추가
def join_study(db: Session, user: User, study_id: int) -> Study:
    study = find_study(db, study_id)
    if study_repo.find_membership(db, study.id, user.id) is not None:
        raise AppError(409, "ALREADY_JOINED", "이미 참가한 모임입니다.")
    # 모집 중이고 자리가 남았을 때만 인원이 늘어납니다. 동시에 마지막 자리를 노려도 한 명만 성공합니다.
    if not study_repo.increase_member_count(db, study.id):
        db.refresh(study)
        reason = "모집이 마감된 모임입니다." if not study.is_open else "정원이 모두 찼습니다."
        raise AppError(409, "CANNOT_JOIN", reason)
    try:
        study_repo.add_membership(db, study.id, user.id)
    except IntegrityError:  # 같은 사람의 동시 요청이 먼저 참가를 기록한 경우
        db.rollback()
        raise AppError(409, "ALREADY_JOINED", "이미 참가한 모임입니다.")
    db.commit()
    db.refresh(study)
    logger.info("모임 참가", extra={"user_id": user.id, "study_id": study.id})
    return study


def leave_study(db: Session, user: User, study_id: int) -> None:
    study = find_study(db, study_id)
    if study.owner_id == user.id:
        raise AppError(409, "OWNER_CANNOT_LEAVE", "개설자는 나갈 수 없습니다. 모임을 삭제하세요.")
    membership = study_repo.find_membership(db, study.id, user.id)
    if membership is None:
        raise AppError(404, "NOT_A_MEMBER", "참가하지 않은 모임입니다.")
    db.delete(membership)
    study_repo.decrease_member_count(db, study.id)
    db.commit()


def list_members(db: Session, user: User, study_id: int) -> list[dict]:
    study = find_study(db, study_id)
    check_can_modify(study, user)  # 명단은 개설자와 관리자만 봅니다.
    return [{"user_id": m.user_id, "email": m.user.email, "joined_at": m.joined_at}
            for m in study_repo.list_members(db, study.id)]


def joined_studies(db: Session, user: User) -> list[Study]:
    return study_repo.list_joined(db, user.id)
