등록, 목록, 상세 API

도서 등록, 목록 조회, 상세 조회를 데이터베이스로 바꿉니다. 이 실습을 마치면 서버를 다시 시작해도 등록한 도서가 남습니다.

실습 목표세션 의존성으로 등록과 조회 API를 데이터베이스와 연결할 수 있습니다.

  1. get_db와 SessionDep 추가

    app/database.py 맨 아래에 get_db를 추가합니다.

    app/database.py
    from sqlalchemy import create_engine
from sqlalchemy.orm import DeclarativeBase, sessionmaker

from app.config import settings

# SQLite는 기본적으로 한 스레드에서만 연결을 쓰게 되어 있어 이 옵션을 끕니다.
connect_args = {"check_same_thread": False} if settings.database_url.startswith("sqlite") else {}

engine = create_engine(settings.database_url, connect_args=connect_args)
SessionLocal = sessionmaker(bind=engine, autoflush=False)


class Base(DeclarativeBase):
    """모든 모델 클래스가 상속하는 기본 클래스입니다."""


def get_db():
    """요청 하나에 세션 하나를 열어 주고, 요청이 끝나면 닫습니다."""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

    app/dependencies.py는 연습용 get_resource를 지우고 SessionDep를 추가합니다.

    app/dependencies.py
    from typing import Annotated

from fastapi import Depends, Query
from sqlalchemy.orm import Session

from app.database import get_db


class Pagination:
    """목록 API가 함께 쓰는 페이지 나누기 값입니다."""

    def __init__(
        self,
        skip: int = Query(default=0, ge=0, description="건너뛸 개수"),
        limit: int = Query(default=10, ge=1, le=100, description="가져올 최대 개수"),
    ):
        self.skip = skip
        self.limit = limit


# 타입과 의존성을 한 이름으로 묶어 두면 여러 함수에서 짧게 쓸 수 있습니다.
PageDep = Annotated[Pagination, Depends()]
SessionDep = Annotated[Session, Depends(get_db)]

  2. 응답 모델에 from_attributes 켜기

    app/schemas.py
    from pydantic import BaseModel, ConfigDict, Field


class BookCreate(BaseModel):
    """등록과 전체 수정(PUT)에 쓰는 입력 모델"""
    title: str = Field(min_length=1, max_length=100, examples=["FastAPI 실전"])
    author: str = Field(min_length=1, max_length=50, examples=["최유나"])
    price: int = Field(ge=0, le=1_000_000, examples=[27000])
    isbn: str | None = Field(default=None, pattern=r"^\d{13}$", examples=["9791100000035"])


class BookUpdate(BaseModel):
    """일부 수정(PATCH)에 쓰는 입력 모델. 모든 항목이 선택입니다."""
    title: str | None = Field(default=None, min_length=1, max_length=100)
    author: str | None = Field(default=None, min_length=1, max_length=50)
    price: int | None = Field(default=None, ge=0, le=1_000_000)
    isbn: str | None = Field(default=None, pattern=r"^\d{13}$")


class BookRead(BaseModel):
    """응답에 쓰는 출력 모델. 여기에 없는 필드는 응답에서 빠집니다."""

    # ORM 객체의 속성(book.title)에서도 값을 읽을 수 있게 합니다.
    model_config = ConfigDict(from_attributes=True)

    id: int
    title: str
    author: str
    price: int
    isbn: str | None = None

  3. 도서 라우터를 데이터베이스로 바꾸기

    app/routers/books.py 전체를 아래 코드로 바꿉니다. 수정(PUT, PATCH)과 삭제(DELETE)는 오후 첫 시간(S4-P12)에 데이터베이스 방식으로 다시 작성하므로 이번에는 빠집니다.

    app/routers/books.py
    from fastapi import APIRouter, status
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.dependencies import PageDep, SessionDep
from app.errors import AppError
from app.models import Book
from app.schemas import BookCreate, BookRead

router = APIRouter(prefix="/books", tags=["도서"])


def find_book(db: Session, book_id: int) -> Book:
    book = db.get(Book, book_id)
    if book is None:
        raise AppError(404, "BOOK_NOT_FOUND", f"{book_id}번 도서를 찾을 수 없습니다.")
    return book


def check_isbn_unique(db: Session, isbn: str | None, exclude_id: int | None = None) -> None:
    if isbn is None:
        return
    found = db.scalar(select(Book).where(Book.isbn == isbn))
    if found is not None and found.id != exclude_id:
        raise AppError(409, "DUPLICATE_ISBN", "이미 등록된 ISBN입니다.")


@router.get("", response_model=list[BookRead], summary="도서 목록 조회")
def list_books(db: SessionDep, page: PageDep, author: str | None = None):
    stmt = select(Book)
    if author is not None:
        stmt = stmt.where(Book.author == author)
    stmt = stmt.order_by(Book.id).offset(page.skip).limit(page.limit)
    return db.scalars(stmt).all()


@router.get("/{book_id}", response_model=BookRead, summary="도서 상세 조회",
            responses={404: {"description": "도서를 찾을 수 없음"}})
def get_book(db: SessionDep, book_id: int):
    return find_book(db, book_id)


@router.post("", response_model=BookRead, status_code=status.HTTP_201_CREATED, summary="도서 등록",
             responses={409: {"description": "이미 등록된 ISBN"}})
def create_book(db: SessionDep, book_in: BookCreate):
    check_isbn_unique(db, book_in.isbn)
    book = Book(**book_in.model_dump())
    db.add(book)      # 저장할 목록에 올립니다.
    db.commit()       # 데이터베이스에 실제로 저장합니다.
    db.refresh(book)  # 데이터베이스가 정한 id를 다시 읽어 옵니다.
    return book

    바뀐 점은 세 가지입니다. 모든 함수가 db: SessionDep로 세션을 받고, BOOKS 목록 대신 세션으로 데이터를 다루며, 찾는 함수와 중복 검사 함수가 세션을 매개변수로 받습니다.
