재발급과 로그아웃 구현

리프레시 토큰을 저장하고, 재발급과 로그아웃 API를 만듭니다. 토큰 교체와 로그아웃 뒤의 동작을 직접 확인하고, 액세스 토큰이 만료되는 순간도 체험합니다.

실습 목표리프레시 토큰으로 액세스 토큰을 재발급하고, 로그아웃하면 재발급을 막을 수 있습니다.

  4. security.py에 함수 두 개 추가

    맨 위에 import hashlib와 import secrets를 추가하고, 맨 아래에 create_refresh_token과 hash_token을 추가합니다.

    app/security.py 전체 펼치기
    app/security.py
    import hashlib
import secrets
from datetime import datetime, timedelta, timezone

import jwt
from pwdlib import PasswordHash

from app.config import settings

# 지금 권장되는 알고리즘(Argon2)으로 비밀번호 해시를 만드는 도구입니다.
password_hash = PasswordHash.recommended()

# 없는 이메일로 로그인할 때 비교용으로 쓰는 가짜 해시입니다.
DUMMY_HASH = password_hash.hash("dummy-password")


def hash_password(password: str) -> str:
    """비밀번호를 해시로 바꿉니다. 같은 비밀번호도 호출할 때마다 다른 결과가 나옵니다."""
    return password_hash.hash(password)


def verify_password(password: str, hashed: str) -> bool:
    """입력한 비밀번호가 저장된 해시와 맞는지 확인합니다."""
    return password_hash.verify(password, hashed)


ALGORITHM = "HS256"  # 비밀 키 하나로 서명하고 검증하는 방식


def create_access_token(user_id: int) -> str:
    """사용자 번호와 만료 시각을 담아 서명한 JWT를 만듭니다."""
    expire = datetime.now(timezone.utc) + timedelta(minutes=settings.access_token_expire_minutes)
    payload = {"sub": str(user_id), "exp": expire}
    return jwt.encode(payload, settings.secret_key, algorithm=ALGORITHM)


def decode_access_token(token: str) -> int | None:
    """토큰이 올바르면 사용자 번호를, 아니면 None을 돌려줍니다."""
    try:
        payload = jwt.decode(
            token,
            settings.secret_key,
            algorithms=[ALGORITHM],              # 이 방식으로 서명한 토큰만 받습니다.
            options={"require": ["exp", "sub"]},  # 만료 시각과 사용자 번호가 꼭 있어야 합니다.
        )
    except jwt.InvalidTokenError:  # 서명이 틀렸거나, 만료되었거나, 형식이 깨진 경우
        return None
    return int(payload["sub"])


def create_refresh_token() -> str:
    """추측할 수 없는 무작위 문자열을 만듭니다. JWT가 아닌 평범한 문자열입니다."""
    return secrets.token_urlsafe(32)


def hash_token(token: str) -> str:
    """리프레시 토큰을 데이터베이스에 저장하고 찾을 때 쓰는 SHA-256 해시를 만듭니다."""
    return hashlib.sha256(token.encode()).hexdigest()

  5. 인증 라우터 완성

    issue_tokens를 추가하고, 로그인 API의 마지막 줄을 바꾸고, 재발급과 로그아웃 API를 추가합니다. 가져오기가 여러 줄 바뀌므로 전체 코드와 비교하며 고칩니다.

    app/routers/auth.py 전체 펼치기
    app/routers/auth.py
    from datetime import datetime, timedelta, timezone
from typing import Annotated

from fastapi import APIRouter, Depends, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.config import settings
from app.dependencies import BEARER, CurrentUserDep, SessionDep
from app.errors import AppError
from app.models import RefreshToken, User
from app.schemas import RefreshRequest, Token, UserCreate, UserRead
from app.security import (DUMMY_HASH, create_access_token, create_refresh_token, hash_password,
                          hash_token, verify_password)

router = APIRouter(prefix="/auth", tags=["인증"])


@router.post("/signup", response_model=UserRead, status_code=status.HTTP_201_CREATED, summary="회원가입",
             responses={409: {"description": "이미 가입된 이메일"}})
def signup(db: SessionDep, user_in: UserCreate):
    email = user_in.email.lower()  # 대문자로 적어도 같은 이메일로 봅니다.
    if db.scalar(select(User).where(User.email == email)) is not None:
        raise AppError(409, "DUPLICATE_EMAIL", "이미 가입된 이메일입니다.")
    user = User(email=email, hashed_password=hash_password(user_in.password))
    db.add(user)
    db.commit()
    db.refresh(user)
    return user


def authenticate(db: Session, email: str, password: str) -> User | None:
    """이메일과 비밀번호가 맞으면 사용자를, 틀리면 None을 돌려줍니다."""
    user = db.scalar(select(User).where(User.email == email.lower()))
    if user is None:
        verify_password(password, DUMMY_HASH)  # 없는 이메일도 비슷한 시간이 걸리게 합니다.
        return None
    if not verify_password(password, user.hashed_password):
        return None
    return user


def issue_tokens(db: Session, user: User) -> dict:
    """액세스 토큰과 리프레시 토큰을 함께 발급합니다. 리프레시 토큰은 해시만 저장합니다."""
    refresh_token = create_refresh_token()
    db.add(RefreshToken(
        user_id=user.id,
        token_hash=hash_token(refresh_token),
        expires_at=datetime.now(timezone.utc) + timedelta(days=settings.refresh_token_expire_days),
    ))
    db.commit()
    return {
        "access_token": create_access_token(user.id),
        "refresh_token": refresh_token,  # 원문은 이 응답에서 한 번만 내보냅니다.
        "token_type": "bearer",
    }


@router.post("/token", response_model=Token, summary="로그인(토큰 발급)",
             responses={401: {"description": "이메일 또는 비밀번호가 틀림"}})
def login(db: SessionDep, form: Annotated[OAuth2PasswordRequestForm, Depends()]):
    # OAuth2 규칙상 입력 칸 이름은 username입니다. 이 앱에서는 이메일을 넣습니다.
    user = authenticate(db, form.username, form.password)
    if user is None:
        raise AppError(401, "INVALID_CREDENTIALS", "이메일 또는 비밀번호가 올바르지 않습니다.", headers=BEARER)
    return issue_tokens(db, user)


@router.get("/me", response_model=UserRead, summary="내 정보 조회",
            responses={401: {"description": "로그인 필요"}})
def read_me(user: CurrentUserDep):
    return user


@router.post("/refresh", response_model=Token, summary="토큰 재발급",
             responses={401: {"description": "리프레시 토큰이 유효하지 않음"}})
def refresh(db: SessionDep, body: RefreshRequest):
    stored = db.scalar(select(RefreshToken).where(
        RefreshToken.token_hash == hash_token(body.refresh_token),
        RefreshToken.revoked.is_(False),
        RefreshToken.expires_at > datetime.now(timezone.utc),
    ))
    if stored is None:
        raise AppError(401, "INVALID_REFRESH_TOKEN", "리프레시 토큰이 유효하지 않거나 만료되었습니다.")
    stored.revoked = True  # 한 번 쓴 리프레시 토큰은 다시 쓰지 못하게 합니다.
    user = db.get(User, stored.user_id)
    return issue_tokens(db, user)  # 새 토큰 저장과 함께 위의 변경도 commit됩니다.


@router.post("/logout", status_code=status.HTTP_204_NO_CONTENT, summary="로그아웃")
def logout(db: SessionDep, body: RefreshRequest):
    stored = db.scalar(select(RefreshToken).where(RefreshToken.token_hash == hash_token(body.refresh_token)))
    if stored is not None:
        stored.revoked = True
        db.commit()
    # 없는 토큰이어도 204를 돌려줍니다. 로그아웃은 몇 번을 해도 결과가 같아야 합니다.

  6. 로그인과 재발급 확인

    서버를 실행하고 POST /auth/token으로 user1이 로그인합니다. 응답에 refresh_token이 함께 오는지 보고, 그 값을 메모장에 복사해 둡니다. 이 값을 A라고 부르겠습니다.

    요청기대하는 결과내 결과
    POST /auth/refresh {"refresh_token": "A"}200, 새 access_token과 A와 다른 새 refresh_token(B)
    POST /auth/refresh {"refresh_token": "A"} 한 번 더401, INVALID_REFRESH_TOKEN. A는 이미 썼습니다.
    POST /auth/refresh {"refresh_token": "abc"}401
    POST /auth/refresh {"refresh_token": ""}422

    요청 본문의 A 자리에는 복사한 실제 값을 넣습니다.

  7. 로그아웃 확인

    요청기대하는 결과내 결과
    POST /auth/logout {"refresh_token": "B"}204, 본문 없음
    같은 요청 한 번 더204
    POST /auth/refresh {"refresh_token": "B"}401. 로그아웃한 토큰으로는 재발급할 수 없습니다.
    B와 함께 받았던 액세스 토큰으로 GET /auth/me200. 액세스 토큰은 만료까지 살아 있습니다.

    마지막 요청은 Swagger UI의 Authorize 대신 curl로 보내면 편합니다(PowerShell이라면 curl.exe).

    curl -i http://127.0.0.1:8000/auth/me -H "Authorization: Bearer 액세스토큰"

    데이터베이스에서 행의 상태도 확인합니다. 로그인, 재발급으로 행이 두 개 생겼고 둘 다 revoked가 1(True)이어야 합니다.

    python -c "import sqlite3; print(sqlite3.connect('books.db').execute('select id, user_id, revoked from refresh_tokens').fetchall())"

  8. 만료 체험하기

    15분을 기다릴 수는 없으니 잠시 1분으로 줄여 봅니다. .env에 한 줄을 추가하고 서버를 다시 시작합니다.

    .env
    ACCESS_TOKEN_EXPIRE_MINUTES=1
    1. Authorize로 user1이 로그인하고 GET /auth/me가 200인지 확인합니다.
    2. 1분 넘게 기다린 뒤 다시 GET /auth/me를 보냅니다. 401, INVALID_TOKEN이 나옵니다.
    3. POST /auth/token으로 받아 둔 리프레시 토큰으로 POST /auth/refresh를 보내 새 액세스 토큰을 받습니다.

    Authorize 창으로 로그인하면 Swagger UI가 액세스 토큰만 기억하고 리프레시 토큰은 보여 주지 않습니다. 3번에 쓸 리프레시 토큰은 POST /auth/token을 직접 호출해 받습니다. 확인이 끝나면 .env에서 추가한 줄을 지우고 서버를 다시 시작합니다.
