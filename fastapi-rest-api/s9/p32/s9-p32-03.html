실습용 취약 코드입니다

이 페이지의 "위험한 코드"는 공격을 직접 확인하려고 일부러 만든 것입니다. 내 PC에서만 실행하고, 확인이 끝나면 바로 안전한 코드로 바꿉니다. Git에 올리거나 서버에 배포하지 않습니다. 공격 실습은 내가 만든 서비스에서만 합니다. 허락받지 않은 남의 서비스에 시도하면 법에 따라 처벌받습니다.

    app/services/loans.py
    """대여 서비스"""
import logging
from datetime import datetime, timezone

from sqlalchemy.orm import Session

from app.errors import AppError
from app.models import Loan, User
from app.repositories import loans as loan_repo
from app.services.books import change_stock, find_book

MAX_ACTIVE_LOANS = 3  # 한 사람이 동시에 빌릴 수 있는 권수
logger = logging.getLogger(__name__)


def borrow(db: Session, user: User, book_id: int) -> Loan:
    book = find_book(db, book_id)
    if loan_repo.find_active(db, user.id, book.id) is not None:
        raise AppError(409, "ALREADY_LOANED", "이미 빌린 도서입니다. 반납한 뒤 다시 빌릴 수 있습니다.")
    if loan_repo.count_active(db, user.id) >= MAX_ACTIVE_LOANS:
        raise AppError(409, "LOAN_LIMIT_EXCEEDED", f"한 번에 {MAX_ACTIVE_LOANS}권까지 빌릴 수 있습니다.")
    if book.stock < 1:
        raise AppError(409, "OUT_OF_STOCK", "남은 재고가 없습니다.")

    # 재고 감소, 재고 기록, 대여 기록을 하나의 트랜잭션으로 묶습니다.
    change_stock(db, book, -1, "대여")
    loan = loan_repo.add(db, Loan(user_id=user.id, book_id=book.id))
    db.commit()
    db.refresh(loan)
    logger.info("도서 대여", extra={"user_id": user.id, "book_id": book.id, "loan_id": loan.id})
    return loan


def find_accessible_loan(db: Session, user: User, loan_id: int) -> Loan:
    """본인이나 관리자가 볼 수 있는 대여 기록만 돌려줍니다. 남의 기록은 없는 것처럼 404로 답합니다."""
    loan = loan_repo.get(db, loan_id)
    if loan is None or (loan.user_id != user.id and user.role != "admin"):
        raise AppError(404, "LOAN_NOT_FOUND", f"{loan_id}번 대여 기록을 찾을 수 없습니다.")
    return loan


def get_loan(db: Session, user: User, loan_id: int) -> Loan:
    return find_accessible_loan(db, user, loan_id)


def return_book(db: Session, user: User, loan_id: int) -> Loan:
    loan = find_accessible_loan(db, user, loan_id)
    if loan.returned_at is not None:
        raise AppError(409, "ALREADY_RETURNED", "이미 반납한 도서입니다.")

    loan.returned_at = datetime.now(timezone.utc)
    change_stock(db, loan.book, 1, "반납")
    db.commit()
    db.refresh(loan)
    logger.info("도서 반납", extra={"user_id": user.id, "book_id": loan.book_id, "loan_id": loan.id})
    return loan


def my_loans(db: Session, user: User) -> list[Loan]:
    return loan_repo.list_by_user(db, user.id)

  4. 관리자 기능 분리

    관리자만 쓰는 기능을 일반 라우터 사이사이에 두면 검사를 빠뜨리기 쉽습니다. /admin 아래로 모으고, 라우터를 만들 때 dependencies로 관리자 검사를 한 번에 겁니다. 이 라우터에 새 API를 추가하면 자동으로 보호됩니다.

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models import User


def list_users(db: Session, skip: int, limit: int) -> list[User]:
    return list(db.scalars(select(User).order_by(User.id).offset(skip).limit(limit)).all())

    app/routers/admin.py
    from fastapi import APIRouter, Depends

from app.dependencies import PageDep, SessionDep, require_admin
from app.repositories import loans as loan_repo
from app.repositories import users as user_repo
from app.schemas import LoanRead, UserRead

# 이 라우터의 모든 API에 관리자 검사를 한 번에 겁니다.
router = APIRouter(prefix="/admin", tags=["관리자"], dependencies=[Depends(require_admin)])


@router.get("/users", response_model=list[UserRead], summary="전체 사용자 목록")
def list_users(db: SessionDep, page: PageDep):
    return user_repo.list_users(db, page.skip, page.limit)


@router.get("/loans", response_model=list[LoanRead], summary="전체 대여 목록")
def list_loans(db: SessionDep, page: PageDep):
    return loan_repo.list_all(db, page.skip, page.limit)
